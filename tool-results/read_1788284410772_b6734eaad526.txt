   101→# ---------- TOC DocTemplate ----------
   102→class TocDocTemplate(SimpleDocTemplate):
   103→    def afterFlowable(self, flowable):
   104→        if hasattr(flowable, 'bookmark_name'):
   105→            level = getattr(flowable, 'bookmark_level', 0)
   106→            text = getattr(flowable, 'bookmark_text', '')
   107→            key = getattr(flowable, 'bookmark_key', '')
   108→            self.notify('TOCEntry', (level, text, self.page, key))

   110→# ---------- Heading helpers ----------
   111→def make_heading(text, style, level=0):
   112→    key = 'h_' + hashlib.md5(text.encode()).hexdigest()[:8]
   113→    p = Paragraph(f'<a name="{key}"/>{text}', style)
   114→    p.bookmark_name = key
   115→    p.bookmark_level = level
   116→    p.bookmark_text = text
   117→    p.bookmark_key = key
   118→    return p
   119→
   120→def h1(text):
   121→    return make_heading(text, H1, level=0)
   122→
   123→def h2(text):

   125→
   126→def h3(text):

   143→def std_table(data, col_ratios, header=True, hAlign='CENTER'):
   144→    """Build a styled table with proper paragraph cells."""
   145→    page_width = A4[0]
   146→    left_margin = 0.75 * inch
   147→    right_margin = 0.75 * inch
   148→    available = page_width - left_margin - right_margin
   149→    col_widths = [r * available for r in col_ratios]
   150→    # Wrap all cells in Paragraph
   151→    wrapped = []
   152→    for row_idx, row in enumerate(data):
   153→        wrapped_row = []
   154→        for cell in row:
   155→            if isinstance(cell, str):
   156→                if row_idx == 0 and header:
   157→                    wrapped_row.append(Paragraph(f'<b>{cell}</b>', TABLE_HEADER_STYLE))
   158→                else:
   159→                    wrapped_row.append(Paragraph(cell, TABLE_CELL_STYLE))
   160→            else:
   161→                wrapped_row.append(cell)
   162→        wrapped.append(wrapped_row)
   163→    t = Table(wrapped, colWidths=col_widths, hAlign=hAlign, repeatRows=1 if header else 0)
   164→    style_cmds = [
   165→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   166→        ('LEFTPADDING', (0, 0), (-1, -1), 7),
   167→        ('RIGHTPADDING', (0, 0), (-1, -1), 7),
   168→        ('TOPPADDING', (0, 0), (-1, -1), 6),
   169→        ('BOTTOMPADDING', (0, 0), (-1, -1), 6),
   170→        ('GRID', (0, 0), (-1, -1), 0.4, BORDER),
   171→    ]
   172→    if header:
   173→        style_cmds += [
   174→            ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   175→            ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   176→        ]
   177→        # Alternating rows
   178→        for i in range(1, len(data)):
   179→            if i % 2 == 0:
   180→                style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_ROW_ODD))
   181→            else:
   182→                style_cmds.append(('BACKGROUND', (0, i), (-1, i), TABLE_ROW_EVEN))
   183→    t.setStyle(TableStyle(style_cmds))
   184→    return t

   186→def callout_box(title, body_text, color=ACCENT, bg=CARD_BG):
   187→    """Decorative callout box for stats and key insights."""
   188→    title_style = ParagraphStyle('CBT', fontName='FreeSerif-Bold', fontSize=11, leading=16,
   189→                                  textColor=color, alignment=TA_LEFT, spaceAfter=4)
   190→    body_style = ParagraphStyle('CBB', fontName='FreeSerif', fontSize=10, leading=15,
   191→                                textColor=TEXT_PRIMARY, alignment=TA_LEFT)
   192→    inner = [Paragraph(title, title_style), Paragraph(body_text, body_style)]
   193→    t = Table([[inner]], colWidths=[A4[0] - 1.5*inch], hAlign='CENTER')
   194→    t.setStyle(TableStyle([
   195→        ('BACKGROUND', (0, 0), (-1, -1), bg),
   196→        ('LINEBEFORE', (0, 0), (0, -1), 3, color),
   197→        ('LEFTPADDING', (0, 0), (-1, -1), 14),
   198→        ('RIGHTPADDING', (0, 0), (-1, -1), 12),
   199→        ('TOPPADDING', (0, 0), (-1, -1), 10),
   200→        ('BOTTOMPADDING', (0, 0), (-1, -1), 10),
   201→    ]))
   202→    return t

   204→def stat_row(stats):
   205→    """Row of stat cards: stats = [(number, label), ...]"""
   206→    cells = []
   207→    for num, label in stats:
   208→        num_style = ParagraphStyle('SN', fontName='FreeSerif-Bold', fontSize=20, leading=24,
   209→                                    textColor=ACCENT, alignment=TA_CENTER)
   210→        lab_style = ParagraphStyle('SL', fontName='FreeSerif', fontSize=9, leading=12,
   211→                                    textColor=TEXT_MUTED, alignment=TA_CENTER)
   212→        cells.append([Paragraph(num, num_style), Paragraph(label, lab_style)])
   213→    col_w = (A4[0] - 1.5*inch) / len(stats)
   214→    t = Table([cells], colWidths=[col_w]*len(stats), hAlign='CENTER')
   215→    t.setStyle(TableStyle([
   216→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   217→        ('TOPPADDING', (0, 0), (-1, -1), 14),
   218→        ('BOTTOMPADDING', (0, 0), (-1, -1), 14),
   219→        ('LINEABOVE', (0, 0), (-1, 0), 2, ACCENT),
   220→        ('LINEBELOW', (0, 0), (-1, -1), 0.5, BORDER),
   221→    ]))
   222→    return t
